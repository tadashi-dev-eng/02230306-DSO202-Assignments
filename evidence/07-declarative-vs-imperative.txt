### Bridge: generate YAML from an imperative command without creating anything
$ kubectl create service clusterip backend-svc --tcp=8080:8080 -n dso202-assignment-01 --dry-run=client -o yaml
apiVersion: v1
kind: Service
metadata:
  labels:
    app: backend-svc
  name: backend-svc
  namespace: dso202-assignment-01
spec:
  ports:
  - name: 8080-8080
    port: 8080
    protocol: TCP
    targetPort: 8080
  selector:
    app: backend-svc
  type: ClusterIP
status:
  loadBalancer: {}
