$ kubectl port-forward -n ingress-nginx svc/ingress-nginx-controller 8443:443 8088:80 &
Forwarding from 127.0.0.1:8443 -> 443

$ H="--resolve tasktracker.local:8443:127.0.0.1 --resolve tasktracker.local:8088:127.0.0.1"   # stands in for /etc/hosts
# 1. TLS: certificate verified against our CA file (no -k)
$ curl -sv $H --cacert ingress/tasktracker.crt https://tasktracker.local:8443/ -o /dev/null 2>&1 | grep -E 'subject:|issuer:|SSL certificate verify|HTTP/2 200'
*  subject: CN=tasktracker.local; O=DSO202
*  issuer: CN=tasktracker.local; O=DSO202
*  SSL certificate verify ok.
< HTTP/2 200 
# 2. ssl-redirect annotation: plain HTTP is redirected to HTTPS
$ curl -sI $H http://tasktracker.local:8088/ | grep -iE '^HTTP|^location'
HTTP/1.1 308 Permanent Redirect
Location: https://tasktracker.local
# 3. / -> frontend-svc
